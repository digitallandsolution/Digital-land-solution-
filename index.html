// ⚠️ আপনার হোয়াটসঅ্যাপ নম্বরটি নিচে ৮৮ সহ পরিবর্তন করুন

const defaultMessage = "হ্যালো Digital Land Solution, আমি আপনাদের ল্যান্ড সার্ভিসের বিষয়ে জানতে চাই।";

// ফি হিসাব করার ফাংশন
function calculateFee() {
const rate = parseFloat(document.getElementById('surveyType').value);
const amount = parseFloat(document.getElementById('landAmount').value);
const resultBox = document.getElementById('totalResult');

if (isNaN(amount) || amount <= 0) {
alert('দয়া করে সঠিক জমির পরিমাণ লিখুন।');
return;
}

const totalCost = rate * amount;
resultBox.innerHTML = `আপনার আনুমানিক মোট সার্ভিস ফি: ${totalCost.toLocaleString('bn-BD')} টাকা।`;
resultBox.style.display = 'block';
}
